cmake_minimum_required(VERSION 3.16)
project(log_viewer_gui LANGUAGES C)
include(FetchContent)
set(RAVEN_BUILD_WEB OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RAVEN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raven GIT_REPOSITORY https://github.com/oscar-soirey/Raven.git GIT_TAG 019ab35d039897ef427dee89092c5602b99f5abf)
FetchContent_MakeAvailable(raven)
file(GLOB CORE CONFIGURE_DEPENDS ../core/*.c)
add_executable(log-viewer-gui gui.c ../app/options.c ${CORE})
set_target_properties(log-viewer-gui PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON)
target_include_directories(log-viewer-gui PRIVATE ../../include)
target_compile_definitions(log-viewer-gui PRIVATE _GNU_SOURCE _DEFAULT_SOURCE _FILE_OFFSET_BITS=64 _XOPEN_SOURCE=700)
target_link_libraries(log-viewer-gui PRIVATE raven::raven pthread)
